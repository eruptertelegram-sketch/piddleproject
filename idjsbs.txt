local function New(c, p, ch)
    local i = Instance.new(c)
    if c=="TextButton" or c=="ImageButton" then i.AutoButtonColor=false end
    if c=="Frame" or c=="TextButton" or c=="ScrollingFrame" or c=="ImageLabel" or c=="ImageButton" then i.BorderSizePixel=0 end

    for _,x in ipairs(ch or {}) do x.Parent=i end
    if p and p.Parent then i.Parent=p.Parent end

local function FormatNumber(v,s)
    if s and s<1 then local d=math.max(0,math.ceil(-math.log10(s))); return string.format("%."..d.."f",v) end
    return tostring(math.floor(v+0.5))
end
local function ResolveImage(src)
    if src==nil or src=="" then return "" end
    if typeof(src)=="number" then return "rbxassetid://"..src end
    if typeof(src)~="string" then return "" end
    if src:find("^rbxasset") or src:find("^rbxthumb") then return src end
    if src:match("^%d+$") then return "rbxassetid://"..src end
    return src
end

    if not id or id == "" then return end
    safeCall(function()
        local s = Instance.new("Sound")
        s.SoundId = id; s.Volume = vol or 1; s.Parent = SoundService; s:Play()
        Debris:AddItem(s, 5)
    end)
end

UIS.MouseIconEnabled = true

local function MakeDraggable(handle, target, getScale, onClick)
    local dragging, startPos, startMouse, moved
    Connect(handle.InputBegan, function(i) if IsPress(i) then dragging=true; moved=0; startMouse=i.Position; startPos=target.Position; safeCall(function() UIS.ModalEnabled=true end) end end)
    Connect(UIS.InputChanged, function(i) if dragging and IsMove(i) then
        local d=i.Position-startMouse; moved=math.max(moved,d.Magnitude)
        local s=getScale and getScale() or 1
        target.Position=UDim2.new(startPos.X.Scale,startPos.X.Offset+d.X/s,startPos.Y.Scale,startPos.Y.Offset+d.Y/s)
    end end)
    Connect(UIS.InputEnded, function(i) if dragging and IsPress(i) then if onClick and moved<5 then onClick() end; dragging=false; safeCall(function() UIS.ModalEnabled=false end) end end)
end

local function Fire(el,...)
    if el.Flag then Library.Flags[el.Flag]=el.Value end
    local ok,err=pcall(el.Callback,...)
    if not ok then warn("[PIDDLE] "..tostring(el.Name)..": "..tostring(err)) end
    for _,fn in ipairs(el._changed) do pcall(fn,...) end
end

        el.Value=Color3.fromHSV(h,s,v)

    end

local Window={}; Window.__index=Window
function Window:_lockDrag(state) self._dragLocked=state end
function Window:_closePopup() if self._activePopup then self._activePopup.Frame.Visible=false; self._activePopup=nil end end
function Window:_applySearch()
    local q=string.lower(self.SearchBox.Text or "")
    local tab=self.CurrentTab; if not tab then return end
    for _,sec in ipairs(tab.Sections) do
        local any=false
        for _,el in ipairs(sec.Elements) do
            local match=q=="" or (el.SearchName~="" and el.SearchName:find(q,1,true)~=nil)
            if el.Type=="Divider" then match=q=="" end
            el.Row.Visible=match; any=any or match
        end
        sec.Frame.Visible=q=="" or any or string.lower(sec.Name):find(q,1,true)~=nil
    end
end
function Window:SelectTab(tab)
    if self.CurrentTab==tab then return end
    self:_closePopup()
    local prev=self.CurrentTab
    if prev then
        prev.Page.Visible=false
        Tween(prev.Bg,TWEEN_FAST,{BackgroundTransparency=1}); Tween(prev.Bar,TWEEN_FAST,{BackgroundTransparency=1})
        Tween(prev.NameLabel,TWEEN_FAST,{TextColor3=Theme.Dim}); Tween(prev.IconRing,TWEEN_FAST,{Color=Theme.Dimmer})
    end
    self.CurrentTab=tab; tab.Page.Visible=true
    Tween(tab.Bg,TWEEN_FAST,{BackgroundTransparency=0.35}); Tween(tab.Bar,TWEEN_FAST,{BackgroundTransparency=0})
    Tween(tab.NameLabel,TWEEN_FAST,{TextColor3=Theme.Text}); Tween(tab.IconRing,TWEEN_FAST,{Color=Theme.Accent})
    self:_applySearch()
end
function Window:SetVisible(state)
    self.Visible=state
    if state then
        self.Root.Visible=true
        Tween(self.UIScale,TWEEN_MED,{Scale=self._baseScale})
        Tween(self.Main,TWEEN_MED,{BackgroundTransparency=0})
        if self.TopImg then self.TopImg.Visible = self._topImageOn ~= false end
    else
        self:_closePopup()
        Tween(self.UIScale,TWEEN_FAST,{Scale=self._baseScale*0.94})
        Tween(self.Main,TWEEN_FAST,{BackgroundTransparency=1})
        if self.TopImg then self.TopImg.Visible = false end
        task.delay(0.16,function() if not self.Visible and self.Root then self.Root.Visible=false end end)
    end
end
function Window:Toggle() self:SetVisible(not self.Visible) end

function Library:Unload()
    if self.Unloaded then return end
    self.Unloaded=true
    for _,c in ipairs(self.Connections) do pcall(function() c:Disconnect() end) end
    table.clear(self.Connections)

    for _,w in ipairs(self.Windows) do pcall(function() w.Root:Destroy() end) end
    table.clear(self.Windows)
    UIS.MouseIconEnabled = true
end

function MV.shape(model)

function MV.exact()
    local out, seen = {}, {}

    for _, n in ipairs({"SummerLobby", "Lobby", "LobbyMap", "Menu"}) do local r = workspace:FindFirstChild(n); if r then inspect(r:FindFirstChild("VotePads", true)) end end
    if #out == 0 then for _, d in ipairs(workspace:GetDescendants()) do if d.Name == "VotePads" or d.Name == "VotePadsFolder" then inspect(d); if #out > 0 then break end end end end

function MV.run(id, slot)
    local function alive() return MV.session == id and MV.running and not Library.Unloaded end
    local entry = MV.entries[slot]; local spot = MV.spot; local startTally = entry and MV.tallyOf(entry) or 0; local last = startTally; local wasEn = entry and entry.info.Enabled; local t0 = os.clock(); local reason = "done"
    while alive() do
        if entry and entry.pad and not entry.pad.Parent then reason = "pad removed"; break end
        if wasEn and not entry.info.Enabled then reason = "voting ended"; break end
        if not MV.infinite and MV.used >= MV.cap then reason = "count reached"; break end
        if os.clock() - t0 > (MV.infinite and 600 or 120) then reason = "timeout"; break end
        local ch = LocalPlayer.Character; local hrp = ch and ch:FindFirstChild("HumanoidRootPart"); local hum = ch and ch:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            hrp.CFrame = CFrame.new(spot); hrp.AssemblyLinearVelocity = Vector3.zero
            if entry and entry.pad then EX.touch(hrp, entry.pad) end
            local limit = (not MV.dupe) and 0.25 or MV.delay; local tt = os.clock()
            while alive() and os.clock() - tt < limit do if MV.dupe and entry and MV.tallyOf(entry) > last then break end; if hrp.Parent and (hrp.Position - spot).Magnitude > 3 then hrp.CFrame = CFrame.new(spot) end; task.wait() end
            if not alive() then break end
            MV.used = MV.used + 1
            if entry then local cur = MV.tallyOf(entry); if cur > last then last = cur end; MV.gain = last - startTally end
            if not MV.dupe then reason = "voted"; break end
            if not MV.infinite and MV.used >= MV.cap then reason = "count reached"; break end
            MV.kill(); local rt = os.clock(); while alive() and not MV.spawned(ch) and os.clock() - rt < 8 do task.wait() end
        else task.wait() end
    end
    if MV.session == id then MV.status = reason .. " (" .. MV.used .. (entry and (" / +" .. MV.gain) or "") .. ")"; MV.finish(true) end
end
function MV.begin()
    if MV.running then return false, "already running" end
    local entries = MV.getEntries(); local entry = entries[MV.pick]; local ch = LocalPlayer.Character; local hrp = ch and ch:FindFirstChild("HumanoidRootPart"); if not hrp then return false, "no character" end
    local spot = MV.spotFor(MV.pick); if not spot then return false, "no platform for slot " .. MV.pick end
    if entry and entry.real and entry.info.Enabled == false then return false, "no active vote pad" end
    MV.session = MV.session + 1; local id = MV.session; MV.running = true; MV.used = 0; MV.gain = 0; MV.status = "voting"; MV.origin = hrp.CFrame; MV.spot = spot; MV.mark = entry and MV.tallyOf(entry) or 0; MV.startTally = MV.mark
    hrp.CFrame = CFrame.new(spot); if entry and entry.pad then EX.touch(hrp, entry.pad) end
    if not MV.dupe then task.delay(0.15, function() if MV.session == id then MV.status = "voted"; MV.finish(true) end end); return true end
    MV.conns.hold = Connect(RunService.Heartbeat, function() if MV.session ~= id or not MV.spot then return end; if entry and entry.info.Enabled == false then MV.status = "voting ended (" .. MV.used .. ")"; MV.finish(true); return end; local c = LocalPlayer.Character; local h = c and c:FindFirstChild("HumanoidRootPart"); if not h then return end; local flat = Vector3.new(h.Position.X - MV.spot.X, 0, h.Position.Z - MV.spot.Z); if flat.Magnitude > 2.5 then h.CFrame = CFrame.new(MV.spot) end end)
    if entry and entry.real then MV.conns.tally = Connect(entry.tally:GetPropertyChangedSignal("Text"), function() if MV.session ~= id or not MV.dupe then return end; local now = MV.tallyOf(entry); if now <= (MV.mark or 0) then MV.mark = now; return end; MV.mark = now; MV.gain = now - (MV.startTally or 0); if not MV.infinite and MV.used >= MV.cap then MV.drop(); task.defer(function() if MV.session == id then MV.status = "count reached (" .. MV.used .. ")"; MV.finish(true) end end); return end; MV.used = MV.used + 1; MV.kill() end) end
    MV.conns.spawn = Connect(LocalPlayer.CharacterAdded, function(char) if MV.session ~= id or not MV.dupe then return end; local h = char:WaitForChild("HumanoidRootPart", 6); if not h or MV.session ~= id or not MV.spot then return end; if entry and entry.info.Enabled == false then return end; if not MV.infinite and MV.used >= MV.cap then MV.status = "count reached (" .. MV.used .. ")"; MV.finish(true); return end; h.CFrame = CFrame.new(MV.spot); if entry and entry.pad then EX.touch(h, entry.pad) end end)
    task.spawn(MV.run, id, MV.pick); return true
end

MusicSec:AddButton({Name="pause", Callback=function() bgSound:Pause() end})
MusicSec:AddButton({Name="stop", Callback=function() bgSound:Stop() end})
MusicSec:AddButton({Name="next", Callback=function() playIdx(curTrack+1) end})
MusicSec:AddButton({Name="prev", Callback=function() playIdx(curTrack-1) end})
